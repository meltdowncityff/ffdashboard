function buildSearchDatalist() { const datalist = document.getElementById('playerSearchList'); if (!datalist) return; datalist.innerHTML = ''; // 1. Add players and draft picks from the database if (window.nflPlayersDb && Array.isArray(window.nflPlayersDb)) { window.nflPlayersDb.forEach(item => { const player = item.player || item; const name = player.name || player.displayName; const pos = player.position; const team = player.maybeTeam || player.team || ''; // Include standard player positions AND draft picks ('PICK') if (['QB', 'RB', 'WR', 'TE', 'K', 'DEF', 'PICK'].includes(pos)) { const option = document.createElement('option'); option.value = name; option.dataset.id = player.id || player.sleeperId; option.dataset.position = pos; option.dataset.team = team; option.dataset.value = item.value || 0; datalist.appendChild(option); } }); } // 2. Ensure standard Dynasty draft picks are always available const currentYear = new Date().getFullYear(); const pickYears = [currentYear, currentYear + 1, currentYear + 2]; const rounds = [1, 2, 3, 4]; pickYears.forEach(year => { rounds.forEach(round => { const ordinal = round === 1 ? '1st' : round === 2 ? '2nd' : round === 3 ? '3rd' : '4th'; const pickName = `${year} ${ordinal} round pick`; // Avoid duplicates if already populated from the database const exists = Array.from(datalist.options).some(opt => opt.value.toLowerCase() === pickName.toLowerCase()); if (!exists) { const option = document.createElement('option'); option.value = pickName; option.dataset.position = 'PICK'; option.dataset.value = getDefaultPickValue(round, year); datalist.appendChild(option); } }); }); } // Helper function to assign a baseline dynasty trade value to fallback picks function getDefaultPickValue(round, year) { const baseValues = { 1: 5500, 2: 2800, 3: 1300, 4: 600 }; let val = baseValues[round] || 1000; // Discount future years slightly const yearDiff = year - new Date().getFullYear(); if (yearDiff > 0) { val *= Math.pow(0.85, yearDiff); } return Math.round(val); }
